INSERT INTO fact_pending_calls (
    ticket_no,
    customer_code,
    branch_id,
    franchise_id,
    product_id,
    technician_id,
    customer_name,
    pincode,
    call_type,
    serial_number,
    machine_status,
    service_type,
    call_book_date,
    scheduled_date,
    pending_reason,
    status,
    repeat_call_flag
)

SELECT
    p.ticket_no,
    p.customer_code,
    b.branch_id,
    f.franchise_id,
    pr.product_id,
    t.technician_id,
    p.customer_name,
    p.pincode,
    p.call_type,
    p.serial_number,
    p.machine_status,
    p.service_type,
    p.call_book_date,
    p.scheduled_date,
    p.pending_reason,
    p.status,
    p.repeat_call_flag

FROM pending_calls p

LEFT JOIN dim_branch b
ON p.branch_name = b.branch_name

LEFT JOIN dim_franchise f
ON p.franchise_name = f.franchise_name

LEFT JOIN dim_product pr
ON p.product = pr.product
AND p.model = pr.model

LEFT JOIN dim_technician t
ON p.technician_name = t.technician_name;

SELECT * FROM fact_pending_calls LIMIT 10;
